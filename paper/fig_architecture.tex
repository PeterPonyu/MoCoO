% ── reusable styles ──────────────────────────────────────────────────────────
\tikzset{
  basenode/.style={

    text width=21mm, align=center, font=\small\sffamily,

  },
  vaebox/.style  ={basenode, fill=cVAE!10,  draw=cVAE!65},
  odebox/.style  ={basenode, fill=cODE!10,  draw=cODE!65},
  mocobox/.style ={basenode, fill=cMoCo!10, draw=cMoCo!65},
  ibbox/.style   ={basenode, fill=cIB!10,   draw=cIB!65},
  iobox/.style   ={basenode, fill=cInput!6, draw=cInput!45, text width=16mm},
  % queue with internal dividers
  queuebox/.style={
    draw=cMoCo!65, fill=cMoCo!7, rounded corners=1.5pt,
    minimum width=18mm, minimum height=7.5mm,
    font=\small\sffamily, align=center,
    inner sep=2pt, line width=0.55pt,
    append after command={
      \pgfextra{
        \foreach \f in {0.2,0.4,0.6,0.8}{

            ($(\tikzlastnode.north west)!\f!(\tikzlastnode.north east)$)
            -- ($(\tikzlastnode.south west)!\f!(\tikzlastnode.south east)$);
        }
      }
    },
  },
  % arrows
  dataflow/.style={
    -{Stealth[length=2mm,width=1.4mm]},

  },
  emaflow/.style={
    -{Stealth[length=1.5mm,width=1mm]},
    line width=0.45pt, densely dotted, color=cMoCo!70,
  },
  % labels

  lossannot/.style={font=\scriptsize\sffamily, text=cLoss!85},
}

% =====================================================================
%  MAIN FLOW (y=0): Input -> Encoder -> z -> (+) -> Decoder -> Output
% =====================================================================

\node[iobox] (input) at (0,0) {
  \textbf{Input}\\[-1pt]
  {\scriptsize $X\!\in\!\mathbb{R}^{N\times G}$}
};

\node[vaebox] (enc) at (2.8,0) {
  \textbf{Encoder}\\[-1pt]
  {\scriptsize $q_\phi(z\!\mid\!x)$}
};

\node[vaebox, text width=12mm, minimum height=8.5mm] (z) at (5.1,0) {
  $\boldsymbol{z}$\\[-1pt]
  {\scriptsize $\mu,\sigma$}
};

% Blend circle (summation point)
\node[draw=black!50, fill=white, circle, inner sep=0pt,
      minimum size=4.5mm, font=\scriptsize\sffamily] (blend) at (7.2,0) {$+$};

\node[vaebox] (dec) at (9.2,0) {
  \textbf{Decoder}\\[-1pt]
  {\scriptsize NB\,/\,ZINB}
};

\node[iobox] (output) at (11.6,0) {
  \textbf{Output}\\[-1pt]
  {\scriptsize $\hat{X}$, $\hat{r}$}
};

% Main-flow arrows
\draw[dataflow] (input) -- node[annot, above, yshift=0.3mm]
  {\scriptsize $\log(1\!+\!x)$} (enc);
\draw[dataflow] (enc)   -- (z);
\draw[dataflow] (z)     -- node[annot, above, yshift=0.3mm]
  {\scriptsize $\alpha_{\!\text{vae}}\!\cdot\! z$} (blend);
\draw[dataflow] (blend) -- node[annot, above, yshift=0.3mm]
  {\scriptsize $z_{\text{blend}}$} (dec);
\draw[dataflow] (dec)   -- (output);

% =====================================================================
%  NEURAL ODE (above main flow, y ~ 1.8)  — compacted
% =====================================================================

\node[odebox, text width=22mm] (ode) at (6.2,1.8) {
  \textbf{Neural ODE}\\[-1pt]
  {\scriptsize $\frac{dz}{dt}\!=\!f_\psi(z,t)$}
};

% Pseudotime from encoder hidden layer
\draw[dataflow, cODE!75]
  (enc.north) -- ++(0,0.4)
  node[annot, above left, xshift=2mm, yshift=-2mm] {\scriptsize $t\!\in\![0,1]$}
  -| (ode.west);

% z -> ODE
\draw[dataflow, cODE!75]
  (z.north) -- ++(0,0.35) -| node[annot, left, yshift=-2mm] {\scriptsize $z$}
  ([xshift=-2mm]ode.south);

% ODE -> blend (z_ode feeds blend circle)
\draw[dataflow, cODE!75]
  (ode.east) -| node[annot, right, yshift=2.5mm]
  {\scriptsize $\alpha_{\!\text{ode}}\!\cdot\! z_{\text{ode}}$} (blend.north);

% ODE losses

\node[lossannot] (velloss) at (6.3,2.8) {$\mathcal{L}_{\text{vel}}$};
\node[lossannot] (crossloss) at (7.4,2.8) {$\mathcal{L}_{\text{cross}}$};
\draw[lossflow] ([xshift=-3mm]ode.north) -- (odeloss);
\draw[lossflow] (ode.north) -- (velloss);
\draw[lossflow] ([xshift=3mm]ode.north) -- (crossloss);

% =====================================================================
%  INFORMATION BOTTLENECK (below blend/decoder area)  — compacted
% =====================================================================

\node[ibbox, text width=26mm] (ib) at (7.5,-1.5) {
  \textbf{Info.\ Bottleneck}\\[-1pt]
  {\scriptsize $z\!\to\! b\!\in\!\mathbb{R}^4\!\to\!\hat{z}$}
};

% z -> IB
\draw[dataflow, cIB!75]
  (z.south) -- ++(0,-0.4) -| (ib.west);

% IB -> decoder
\draw[dataflow, cIB!75]
  (ib.north) -- ++(0,0.3) -|
  node[annot, above, near end, yshift=0.3mm] {\scriptsize $\hat{z}$}
  (dec.south);

\node[lossannot] (ibloss) at (7.5,-2.5) {$\mathcal{L}_{\text{irecon}}$};
\draw[lossflow] (ib.south) -- (ibloss);

% =====================================================================
%  LOSSES near z and decoder
% =====================================================================

% KL divergence (above z, offset left)
\node[lossannot] (klloss) at (3.8,1.05) {$\beta\!\cdot\! D_{\text{KL}}$};

% Reconstruction loss (above decoder)
\node[lossannot] (reconloss) at (9.6,1.05) {$\mathcal{L}_{\text{recon}}$};
\draw[lossflow] (dec.north) -- (reconloss);

% Disentanglement losses (below main flow, left side)
\node[lossannot, text width=28mm, align=center] (disloss) at (3.8,-1.1) {

};

};

};

% Projection heads

};
\node[mocobox, text width=16mm] (projk) at (5.3,-4.6) {
  \textbf{Proj.\ $h_k$}\\[-1pt]
  {\scriptsize MLP+BN}
};

% InfoNCE
\node[mocobox, text width=15mm, minimum height=9.5mm] (infonce) at (7.8,-3.3) {

};

  node[annot, above] {\scriptsize $\mu_k$} (projk);

  node[annot, right, xshift=0.5mm] {\scriptsize $k^-$}
